# tecnigrafo: the Qt desktop application. The only module allowed to
# depend on Qt (docs/adr/0001).

find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()

# Everything but main(): shared by the application and its Qt tests.
qt_add_library(tg_app_ui STATIC
    src/action_registry.cpp
    src/action_registry.hpp
    src/app_commands.cpp
    src/app_commands.hpp
    src/ui_builder.cpp
    src/ui_builder.hpp
    src/builtin_actions.cpp
    src/builtin_actions.hpp
    src/cad_view.cpp
    src/cad_view.hpp
    src/command_line.cpp
    src/command_line.hpp
    src/main_window.cpp
    src/main_window.hpp
    src/recent_files.cpp
    src/recent_files.hpp
    src/qt_painter.cpp
    src/qt_painter.hpp
    src/sample_drawing.cpp
    src/sample_drawing.hpp
    src/script_session.cpp
    src/script_session.hpp
)
qt_add_resources(tg_app_ui fonts
    PREFIX "/tg/fonts"
    BASE "${PROJECT_SOURCE_DIR}/resources/fonts"
    FILES "${PROJECT_SOURCE_DIR}/resources/fonts/LiberationSans-Regular.ttf")

target_include_directories(tg_app_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
# Tecnigrafo::script writes the report of the control mode at level 2
# (--script, docs/adr/0011).
target_link_libraries(tg_app_ui PUBLIC
    Tecnigrafo::cmd Tecnigrafo::render Tecnigrafo::script tg_saribbon Qt6::Widgets)
target_compile_definitions(tg_app_ui PRIVATE QT_DISABLE_DEPRECATED_UP_TO=0x060400)
tg_configure_target(tg_app_ui)

qt_add_executable(tecnigrafo WIN32 MACOSX_BUNDLE
    src/main.cpp
)

target_link_libraries(tecnigrafo PRIVATE tg_app_ui Qt6::Widgets)
target_compile_definitions(tecnigrafo PRIVATE
    TG_VERSION="${PROJECT_VERSION}"
    QT_DISABLE_DEPRECATED_UP_TO=0x060400)
tg_configure_target(tecnigrafo)

# Code generated by Qt's moc is not ours to lint (CMake >= 3.27 honours this).
set_source_files_properties(
    "${CMAKE_CURRENT_BINARY_DIR}/tecnigrafo_autogen/mocs_compilation.cpp"
    "${CMAKE_CURRENT_BINARY_DIR}/tg_app_ui_autogen/mocs_compilation.cpp"
    PROPERTIES SKIP_LINTING ON)

set_target_properties(tecnigrafo PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "Tecnigrafo"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.tecnigrafo.Tecnigrafo"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")

# Smoke test: start the application without a display, render the main
# window to a PNG and exit. Catches crashes at start-up on every platform.
if(TG_BUILD_TESTS)
    add_test(NAME app_smoke
        COMMAND tecnigrafo --screenshot "${CMAKE_BINARY_DIR}/smoke.png")
    set_tests_properties(app_smoke PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
        TIMEOUT 60)

endif()

install(TARGETS tecnigrafo
    BUNDLE DESTINATION .
    RUNTIME DESTINATION bin)
